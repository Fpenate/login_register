SELECT id, location_code, location_type
	FROM (
	    SELECT id, location_code, location_type,
	           ROW_NUMBER() OVER (PARTITION BY location_type ORDER BY id DESC) as rn
	    FROM conformal_inventory_db.locatios
	) as sub
	WHERE rn = 1
